+import unittest
+
+from intervals import merge
+
+
+class MergeTests(unittest.TestCase):
+    def test_touching_intervals_remain_separate(self):
+        self.assertEqual(merge([(1, 3), (3, 5)]), [(1, 3), (3, 5)])
+
+    def test_unsorted_overlaps_merge_but_touching_intervals_do_not(self):
+        intervals = [(8, 10), (3, 6), (6, 8), (1, 4)]
+        original = intervals.copy()
+
+        self.assertEqual(merge(intervals), [(1, 6), (6, 8), (8, 10)])
+        self.assertEqual(intervals, original)
+
+    def test_nested_and_duplicate_intervals_merge(self):
+        self.assertEqual(merge([(2, 3), (1, 5), (1, 5)]), [(1, 5)])
+
+    def test_overlap_can_bridge_touching_intervals(self):
+        self.assertEqual(merge([(3, 5), (2, 4), (1, 3)]), [(1, 5)])
+
+    def test_empty_input(self):
+        self.assertEqual(merge([]), [])
+
+    def test_single_interval(self):
+        self.assertEqual(merge([(-3, -1)]), [(-3, -1)])
