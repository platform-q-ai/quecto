+import unittest
+
+from search import find
+
+
+class SearchTests(unittest.TestCase):
+    def test_default_and_none_preserve_matching_order_and_duplicates(self):
+        names = ["Tea", "coffee", "teapot", "Tea", "ICED TEA"]
+        expected = ["Tea", "teapot", "Tea", "ICED TEA"]
+        self.assertEqual(find(names, "TEA"), expected)
+        self.assertEqual(find(names, "TEA", limit=None), expected)
+
+    def test_limit_counts_matches_including_duplicates(self):
+        names = ["coffee", "Tea", "water", "Tea", "teapot"]
+        for limit, expected in [
+            (0, []),
+            (1, ["Tea"]),
+            (2, ["Tea", "Tea"]),
+            (3, ["Tea", "Tea", "teapot"]),
+            (10, ["Tea", "Tea", "teapot"]),
+        ]:
+            with self.subTest(limit=limit):
+                self.assertEqual(find(names, "tea", limit=limit), expected)
+
+    def test_negative_limit_raises_even_without_matches(self):
+        for names in ([], ["coffee"], ["Tea"]):
+            with self.subTest(names=names):
+                with self.assertRaises(ValueError):
+                    find(names, "tea", limit=-1)
+
+    def test_empty_query_and_no_matches(self):
+        self.assertEqual(find(["B", "A", "B"], "", limit=2), ["B", "A"])
+        self.assertEqual(find(["Tea"], "coffee", limit=2), [])
+        self.assertEqual(find([], "tea", limit=2), [])
+
+    def test_unicode_casefold(self):
+        self.assertEqual(find(["Straße", "STRASSE"], "strasse", limit=1), ["Straße"])
+
+    def test_inputs_unchanged(self):
+        names = ["Tea", "coffee", "Tea", "teapot"]
+        original = names.copy()
+        query = "TEA"
+        for limit in (None, 0, 1, 10, -1):
+            with self.subTest(limit=limit):
+                if limit == -1:
+                    with self.assertRaises(ValueError):
+                        find(names, query, limit=limit)
+                else:
+                    result = find(names, query, limit=limit)
+                    self.assertIsNot(result, names)
+                self.assertEqual(names, original)
+                self.assertEqual(query, "TEA")
